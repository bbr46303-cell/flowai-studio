const generateBtn = document.querySelector('button'); // Aapka generate button generateBtn.addEventListener('click', async () => { const promptText = document.querySelector('textarea').value; generateBtn.innerText = "Processing..."; generateBtn.disabled = true; try { // 1. Video job start karein const res = await fetch('/api/generate-video', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: promptText }) }); const data = await res.json(); if (!data.id) throw new Error(data.error || "Failed to start generation"); // 2. Status check polling (Har 5 sec) const interval = setInterval(async () => { const statusRes = await fetch(`/api/video-status/${data.id}`); const statusData = await statusRes.json(); if (statusData.status === 'succeeded') { clearInterval(interval); generateBtn.innerText = "Generate Video"; generateBtn.disabled = false; alert("Video Ready! URL: " + statusData.output); } else if (statusData.status === 'failed') { clearInterval(interval); generateBtn.innerText = "Generate Video"; generateBtn.disabled = false; alert("Generation Failed!"); } }, 5000); } catch (err) { alert("Error: " + err.message); generateBtn.innerText = "Generate Video"; generateBtn.disabled = false; } });
